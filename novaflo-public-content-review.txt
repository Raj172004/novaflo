import "server-only";

import { FieldValue } from "firebase-admin/firestore";
import { adminDb } from "@/lib/firebase-admin";

export type ContentType = "service" | "insight";

export interface SiteContent {
  id: string;
  type: ContentType;
  slug: string;
  title: string;
  excerpt: string;
  body: string;
  published: boolean;
  updatedAt: string;
  createdAt: string;
}

const collection = () => adminDb.collection("siteContent");

function normalizeDocument(
  id: string,
  data: FirebaseFirestore.DocumentData,
): SiteContent {
  const dateValue = (value: unknown): string => {
    if (typeof value === "string") return value;
    if (
      value &&
      typeof value === "object" &&
      "toDate" in value &&
      typeof value.toDate === "function"
    ) {
      return value.toDate().toISOString();
    }
    return "";
  };

  return {
    id,
    type: data.type,
    slug: data.slug,
    title: data.title,
    excerpt: data.excerpt ?? "",
    body: data.body ?? "",
    published: data.published === true,
    createdAt: dateValue(data.createdAt),
    updatedAt: dateValue(data.updatedAt),
  };
}

export async function listContent(): Promise<SiteContent[]> {
  const snapshot = await collection().get();

  return snapshot.docs
    .map((doc) => normalizeDocument(doc.id, doc.data()))
    .sort((a, b) => b.updatedAt.localeCompare(a.updatedAt));
}

export async function getPublishedContent(
  type: ContentType,
): Promise<SiteContent[]> {
  const snapshot = await collection()
    .where("type", "==", type)
    .where("published", "==", true)
    .get();

  return snapshot.docs.map((doc) => normalizeDocument(doc.id, doc.data()));
}

export async function saveContent(
  input: Omit<SiteContent, "id" | "createdAt" | "updatedAt">,
  id?: string,
): Promise<string> {
  const now = FieldValue.serverTimestamp();
  const data = {
    type: input.type,
    slug: input.slug,
    title: input.title,
    excerpt: input.excerpt,
    body: input.body,
    published: input.published,
    updatedAt: now,
  };

  if (id) {
    const ref = collection().doc(id);
    const existing = await ref.get();

    if (!existing.exists) {
      throw new Error("CONTENT_NOT_FOUND");
    }

    await ref.set(data, { merge: true });
    return id;
  }

  const ref = collection().doc();
  await ref.set({
    ...data,
    createdAt: now,
  });

  return ref.id;
}

export async function deleteContent(id: string): Promise<boolean> {
  const ref = collection().doc(id);
  const existing = await ref.get();

  if (!existing.exists) return false;

  await ref.delete();
  return true;
}

function formatDate(value: string) {
  if (!value) return "Recently published";

  const date = new Date(value);
  if (Number.isNaN(date.getTime())) return "Recently published";

  return new Intl.DateTimeFormat("en", {
    year: "numeric",
    month: "short",
    day: "numeric",
  }).format(date);
}

function readingTime(body: string) {
  const words = body.trim().split(/\s+/).filter(Boolean).length;
  return `${Math.max(1, Math.ceil(words / 200))} min read`;
}

function getParagraphs(body: string) {
  return body
    .split(/\n\s*\n/)
    .map((paragraph) => paragraph.trim())
    .filter(Boolean);
}

function getReadingTime(body: string) {
  const words = body.trim().split(/\s+/).filter(Boolean).length;
  return `${Math.max(1, Math.ceil(words / 200))} min read`;
}

function getPublishedDate(createdAt: string, updatedAt: string) {
  const value = createdAt || updatedAt;
  if (!value) return new Date().toISOString();

  const date = new Date(value);
  return Number.isNaN(date.getTime())
    ? new Date().toISOString()
    : date.toISOString();
}

export async function generateMetadata({
  params,
}: {
  params: Promise<{ slug: string }>;
}): Promise<Metadata> {
  const { slug } = await params;

    return pageMetadata({
      title: post.title,
      description: post.excerpt,
      path: `/insights/${post.slug}`,
    });
  } catch {
    return {};
  }
}
